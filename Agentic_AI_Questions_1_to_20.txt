Question 1: Create a simple Python agent using LangChain that uses the ReAct method to calculate the total cost of items purchased by a user. Use FAISS memory to store previous purchases and retrieve them for a future query.

SOURCE CODE:
Filename: question_01.py
import os, ast, operator
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_core.documents import Document
from langchain_core.tools import tool
from langchain.agents import create_agent

DB = "q01_faiss"
emb = OpenAIEmbeddings()
try: memory = FAISS.load_local(DB, emb, allow_dangerous_deserialization=True)
except Exception: memory = None

@tool
def calculate_total(items: str) -> str:
    """Calculate purchase total. Input is a Python-style list of (item, quantity, unit_price)."""
    rows = ast.literal_eval(items)
    total = sum(float(q) * float(p) for _, q, p in rows)
    return f"Purchase: {rows}; total={total:.2f}"

model = ChatOpenAI(model=os.getenv("OPENAI_MODEL", "gpt-4o-mini"), temperature=0)
agent = create_agent(model, tools=[calculate_total], system_prompt="Use the calculate_total tool for arithmetic. Answer briefly.")
query = input("Enter items as [('pen',2,10),('book',1,50)] or ask about previous purchases: ")
if "previous" in query.lower() and memory:
    print("\n".join(d.page_content for d in memory.similarity_search(query, k=3)))
else:
    result = agent.invoke({"messages":[{"role":"user","content":query}]})
    answer = result["messages"][-1].content
    print(answer)
    if memory is None: memory = FAISS.from_documents([Document(page_content=f"User query: {query}\nResult: {answer}")], emb)
    else: memory.add_documents([Document(page_content=f"User query: {query}\nResult: {answer}")])
    memory.save_local(DB)

INPUT FORMAT:
Enter items as a Python-style list of (item, quantity, unit_price), or ask about previous purchases. Example: [('pen',2,10),('book',1,50)].

OUTPUT FORMAT:
A purchase summary and total, or matching saved purchase records for a previous-purchase query. Example: 'total=70.00'.

Question 3: Build a simple agent using LangGraph that uses the Tree-of-Thought method to generate three possible solutions for selecting a suitable study timetable. Use FAISS to store the user's previous timetable preferences.

SOURCE CODE:
Filename: question_03.py
import os
from typing import TypedDict
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_core.documents import Document
from langgraph.graph import StateGraph, START, END

DB="q03_faiss"
emb=OpenAIEmbeddings()
try: memory=FAISS.load_local(DB,emb,allow_dangerous_deserialization=True)
except Exception: memory=None
llm=ChatOpenAI(model=os.getenv("OPENAI_MODEL","gpt-4o-mini"),temperature=0)
class State(TypedDict):
    request:str
    past:str
    result:str
def branch(s):
    past=memory.similarity_search(s["request"],k=3) if memory else []
    prompt=f"Tree-of-Thought: independently propose exactly three distinct study timetables, compare each against the constraints, and recommend one. Do not reveal private chain-of-thought; provide concise criteria and conclusions.\nPast preferences: {[d.page_content for d in past]}\nRequest: {s['request']}"
    return {"past":str([d.page_content for d in past]),"result":llm.invoke(prompt).content}
g=StateGraph(State); g.add_node("explore",branch); g.add_edge(START,"explore"); g.add_edge("explore",END)
request=input("Study subjects, available times, and constraints: ")
result=g.compile().invoke({"request":request,"past":"","result":""})["result"]
print(result)
if memory is None: memory=FAISS.from_documents([Document(page_content=f"{request}\n{result}")],emb)
else: memory.add_documents([Document(page_content=f"{request}\n{result}")])
memory.save_local(DB)

INPUT FORMAT:
Enter subjects, available times, and timetable constraints. Example: 'Maths, AI, DBMS; weekdays 5–8 PM; no study after 7 PM Friday'.

OUTPUT FORMAT:
Three timetable options, concise comparison, and a recommended feasible option.

Question 19: Create a Python agent using LangChain that uses the ReAct method to plan a simple budget for a college student's monthly expenses. Use ChromaDB to remember previous expense categories.

SOURCE CODE:
Filename: question_19.py
import os
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_chroma import Chroma
from langchain_core.documents import Document
from langchain_core.tools import tool
from langchain.agents import create_agent

emb=OpenAIEmbeddings(); memory=Chroma(collection_name="expense_categories",embedding_function=emb,persist_directory="./q19_chroma")
llm=ChatOpenAI(model=os.getenv("OPENAI_MODEL","gpt-4o-mini"),temperature=0)
@tool
def calculate_budget(income_and_expenses: str) -> str:
    """Calculate remaining budget from JSON with income and expenses numeric values."""
    import json
    data=json.loads(income_and_expenses)
    income=float(data["income"])
    expenses={k:float(v) for k,v in data["expenses"].items()}
    return json.dumps({"income":income,"expenses":expenses,"total_expenses":round(sum(expenses.values()),2),"remaining":round(income-sum(expenses.values()),2)})
agent=create_agent(llm,tools=[calculate_budget],system_prompt="Use the budget calculation tool when numeric income and expense data are provided. Make a realistic budget and do not invent prices.")
req=input("Monthly income and expenses (include amounts), or describe budget needs: ")
past=memory.similarity_search(req,k=3)
prompt=f"Use a ReAct-style approach to create a practical student monthly budget, separate needs and wants, and identify savings opportunities. Previous expense categories: {[d.page_content for d in past]}. Request: {req}"
result=agent.invoke({"messages":[{"role":"user","content":prompt}]})
out=result["messages"][-1].content; print(out)
memory.add_documents([Document(page_content=f"Expense categories/request: {req}\nBudget: {out}")])

INPUT FORMAT:
Enter monthly income and expenses or describe budget needs. Example: 'Income 8000; hostel 3000; food 2000; travel 800; books 500'.

OUTPUT FORMAT:
A student budget with categories, total expenses, remaining amount when numeric data is provided, and suggestions.
